cmake_minimum_required(VERSION 3.16)
project(resona-opus-config NONE)

function(check_neon system processor architectures may_have presume)
  set(CMAKE_SYSTEM_NAME "${system}")
  set(CMAKE_SYSTEM_PROCESSOR "${processor}")
  set(CMAKE_OSX_ARCHITECTURES "${architectures}")
  set(OPUS_MAY_HAVE_NEON ON CACHE BOOL "" FORCE)
  set(OPUS_PRESUME_NEON OFF CACHE BOOL "" FORCE)
  set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS "")
  include("${CMAKE_CURRENT_LIST_DIR}/../../native/opus/apple-neon.cmake")
  if(NOT OPUS_MAY_HAVE_NEON STREQUAL may_have OR
     NOT OPUS_PRESUME_NEON STREQUAL presume)
    message(FATAL_ERROR "Unexpected NEON configuration for ${system}/${processor}/${architectures}")
  endif()
  get_directory_property(definitions COMPILE_DEFINITIONS)
  if(presume STREQUAL "ON")
    if(NOT "OPUS_ARM_MAY_HAVE_NEON" IN_LIST definitions OR
       NOT "OPUS_ARM_MAY_HAVE_NEON_INTR" IN_LIST definitions)
      message(FATAL_ERROR "NEON declarations missing")
    endif()
  elseif(definitions)
    message(FATAL_ERROR "NEON definitions leaked to another target")
  endif()
endfunction()

check_neon(Darwin arm64 "" OFF ON)
check_neon(Darwin aarch64 "" OFF ON)
check_neon(Darwin x86_64 arm64 OFF ON)
check_neon(Darwin x86_64 "" ON OFF)
check_neon(Darwin arm64 x86_64 ON OFF)
check_neon(Linux aarch64 "" ON OFF)
check_neon(Windows AMD64 "" ON OFF)
set(OPUS_DISABLE_INTRINSICS ON)
check_neon(Darwin arm64 "" ON OFF)
message(STATUS "Opus target configuration checks passed")
